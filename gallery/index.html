    Kernel gallery¶

    Every featured kernel below is shown in Parakeet's standard recurring styles so you can compare how each idiom compiles:

    Suffix Style

    1. Pairwise fusion and automatic tiling¶

    Exact nearest neighbor¶

    Find the closest reference vector among R references for each of Q queries in Rᴰ. Naive NumPy allocates a Q × R distance matrix (64 MiB at Q = 1,024, R = 16,384); Parakeet streams 16 KiB tiles in L1/SRAM and keeps NumPy's exact first-index rule on ties and NaNs.

    @par.jit
def nearest_numpy(queries, references):
  delta = queries[:, None, :] - references[None, :, :]
  return np.argmin(np.sum(delta**2, axis=2), axis=1)

    def squared_distance(query, reference):
  return np.sum((query - reference) ** 2)



@par.jit
def nearest_comp(queries, references):
  return np.array([np.argmin([squared_distance(q, r) for r in references]) for q in queries])

    def squared_distance(query, reference):
  return np.sum((query - reference) ** 2)



@par.jit
def nearest_loop(queries, references):
  out = np.empty(len(queries), dtype=np.int64)
  for i in range(len(queries)):
    best_idx = 0
    best_dist = squared_distance(queries[i], references[0])
    for j in range(1, len(references)):
      dist = squared_distance(queries[i], references[j])
      if (np.isnan(dist) or dist < best_dist) and not np.isnan(best_dist):
        best_dist = dist
        best_idx = j
    out[i] = best_idx
  return out

    def squared_distance(query, reference):
  return np.sum((query - reference) ** 2)



class Winner(NamedTuple):
  distance: np.float32
  index: np.int64


def first_minimum(left, right):
  # Keep the first NaN; among finite values, replace only on strict improvement.
  right_wins = np.isnan(right.distance) or right.distance < left.distance
  return right if right_wins and not np.isnan(left.distance) else left


@par.jit
def nearest_par(queries, references):
  scores = par.allpairs(squared_distance, queries, references)
  return par.fold(Winner, first_minimum, scores, axis=1, indexed=True, emit=lambda winner: winner.index)

    nearest_numpy, nearest_comp, and nearest_par all lower to the same tiled pairwise schedule on CPU, Metal, and CUDA. Compare timings and scratch memory.

    Inverse-distance smoothing¶

    Smooth reference targets with weight w[i, j] = 1 / (1 + ‖q[i] − r[j]‖²) in a single tiled pass without materializing the Q × R weight matrix:

    @par.jit
def smooth_numpy(queries, references, observations):
  values = observations.reshape(references.shape[0])
  delta = queries[:, None, :] - references[None, :, :]
  weights = 1.0 / (1.0 + np.sum(delta**2, axis=2))
  return np.sum(weights * values, axis=1) / np.sum(weights, axis=1)

    def squared_distance(query, reference):
  return np.sum((query - reference) ** 2)



@par.jit
def smooth_comp(queries, references, observations):
  values = observations.reshape(references.shape[0])

  def smooth_one(q):
    w = np.array([1.0 / (1.0 + squared_distance(q, r)) for r in references])
    return np.sum(w * values) / np.sum(w)

  return np.array([smooth_one(q) for q in queries])

    def squared_distance(query, reference):
  return np.sum((query - reference) ** 2)



@par.jit
def smooth_loop(queries, references, observations):
  values = observations.reshape(references.shape[0])
  out = np.empty(len(queries), dtype=queries.dtype)
  for i in range(len(queries)):
    w_sum = 0.0
    wv_sum = 0.0
    for j in range(len(references)):
      w = 1.0 / (1.0 + squared_distance(queries[i], references[j]))
      w_sum += w
      wv_sum += w * values[j]
    out[i] = wv_sum / w_sum
  return out

    def squared_distance(query, reference):
  return np.sum((query - reference) ** 2)



class WeightedSum(NamedTuple):
  weight: np.float32
  weighted_value: np.float32


def add_sums(left, right):
  return WeightedSum(left.weight + right.weight, left.weighted_value + right.weighted_value)


@par.jit
def smooth_par(queries, references, observations):
  values = observations.reshape(references.shape[0])
  scores = par.allpairs(squared_distance, queries, references)

  def contribution(distance_squared, ref_index):
    weight = np.float32(1) / (np.float32(1) + distance_squared)
    return WeightedSum(weight, weight * values[ref_index])

  return par.fold(contribution, add_sums, scores, axis=1, indexed=True, emit=lambda t: t.weighted_value / t.weight)

    Gaussian kernel regression¶

    Nadaraya-Watson smoothing weights references by exp(-gamma * ‖q[i] − x[j]‖²). Law 1 rescaled-monoid tiling streams tiles without allocating an N × M weight matrix:

    @par.jit
def rbf_regression_numpy(x_query, x_train, y_train, neg_gamma):
  d2 = np.sum((x_query[:, None, :] - x_train[None, :, :]) ** np.float32(2.0), axis=-1)
  s = d2 * neg_gamma
  e = np.exp(s - np.max(s, axis=-1, keepdims=True))
  return (e / np.sum(e, axis=-1, keepdims=True)) @ y_train

    @par.jit
def rbf_regression_helpers(x_query, x_train, y_train, neg_gamma):
  d2 = np.sum((x_query[:, None, :] - x_train[None, :, :]) ** np.float32(2.0), axis=-1)
  return par.nn.softmax(d2 * neg_gamma, axis=-1) @ y_train

    @par.jit
def rbf_regression_comp(x_query, x_train, y_train, neg_gamma):
  def predict_one(q):
    s = np.array([squared_distance(q, x) * neg_gamma for x in x_train])
    e = np.exp(s - np.max(s))
    return (e / np.sum(e)) @ y_train

  return np.array([predict_one(q) for q in x_query])

    @par.jit
def rbf_regression_loop(x_query, x_train, y_train, neg_gamma):
  n_q = len(x_query)
  n_tr = len(x_train)
  out = np.zeros((n_q, y_train.shape[1]), dtype=x_query.dtype)
  for i in range(n_q):
    m = -np.inf
    for j in range(n_tr):
      s = squared_distance(x_query[i], x_train[j]) * neg_gamma
      if s > m:
        m = s
    denom = 0.0
    for j in range(n_tr):
      w = np.exp(squared_distance(x_query[i], x_train[j]) * neg_gamma - m)
      denom += w
      out[i] += w * y_train[j]
    out[i] /= denom
  return out

    @par.jit
def rbf_regression_par(x_query, x_train, y_train, neg_gamma):
  s = par.allpairs(squared_distance, x_query, x_train) * neg_gamma
  e = np.exp(s - np.max(s, axis=-1, keepdims=True))
  return (e / np.sum(e, axis=-1, keepdims=True)) @ y_train

    2. Histograms and scatter-add fusion¶

    Histograms and tree splits¶

    Gradient-boosted trees accumulate gradients and Hessians into feature bins and pick the split with maximum gain. All four styles below share the same gain helper and lower to O(N) IndexScatter, fusing both histograms into one thread-privatized pass:

    def _best_split_from_histograms(g_hist, h_hist, nbins, reg_lambda):
  g_left = np.cumsum(g_hist)
  h_left = np.cumsum(h_hist)
  g_right = g_left[nbins - 1] - g_left
  h_right = h_left[nbins - 1] - h_left
  gain = (g_left * g_left) / (h_left + reg_lambda) + (g_right * g_right) / (h_right + reg_lambda)
  best_bin = np.argmax(gain)
  return best_bin, gain[best_bin]

    @par.jit
def gbdt_split_numpy(bin_ids, grads, hess, nbins, reg_lambda):
  g_hist = np.zeros(nbins, dtype=np.float32)
  h_hist = np.zeros(nbins, dtype=np.float32)
  np.add.at(g_hist, bin_ids, grads)
  np.add.at(h_hist, bin_ids, hess)
  return _best_split_from_histograms(g_hist, h_hist, nbins, reg_lambda)

    Inverts [np.sum(grads[bin_ids == b]) for b in range(nbins)] automatically from O(nbins · N) to one O(N) fused histogram pass:

    @par.jit
def gbdt_split_comp(bin_ids, grads, hess, nbins, reg_lambda):
  g_hist = np.array([np.sum(grads[bin_ids == b]) for b in range(nbins)])
  h_hist = np.array([np.sum(hess[bin_ids == b]) for b in range(nbins)])
  return _best_split_from_histograms(g_hist, h_hist, nbins, reg_lambda)

    Lifts independent indexed += updates into one fused parallel histogram:

    @par.jit
def gbdt_split_loop(bin_ids, grads, hess, nbins, reg_lambda):
  g_hist = np.zeros(nbins, dtype=np.float32)
  h_hist = np.zeros(nbins, dtype=np.float32)
  for i in range(len(bin_ids)):
    b = bin_ids[i]
    g_hist[b] += grads[i]
    h_hist[b] += hess[i]
  return _best_split_from_histograms(g_hist, h_hist, nbins, reg_lambda)

    @par.jit
def gbdt_split_par(bin_ids, grads, hess, nbins, reg_lambda):
  g_hist = par.scatter_add(nbins, bin_ids, grads)
  h_hist = par.scatter_add(nbins, bin_ids, hess)
  return _best_split_from_histograms(g_hist, h_hist, nbins, reg_lambda)

    3. Custom semirings with index operators¶

    Min-plus matrix product¶

    Tropical matrix multiplication replaces (*, +) with (+, min): C[i, j] = min_r (A[i, r] + B[r, j]). Writing the index space explicitly with par.imap and par.ireduce(..., combine=np.minimum) avoids an m × n × k broadcast tensor and triggers cache and register tiling:

    @par.jit
def tropical_min_plus_numpy(a, b):
  return np.min(a[:, :, None] + b[None, :, :], axis=1)

    @par.jit
def tropical_min_plus_comp(a, b):
  return np.array([[np.min(a[i] + b[:, j]) for j in range(b.shape[1])] for i in range(a.shape[0])])

    @par.jit
def tropical_min_plus_loop(a, b):
  m, k = a.shape
  _, n = b.shape
  out = np.empty((m, n), dtype=a.dtype)
  for i in range(m):
    for j in range(n):
      best = a[i, 0] + b[0, j]
      for r in range(1, k):
        val = a[i, r] + b[r, j]
        if val < best:
          best = val
      out[i, j] = best
  return out

    @par.jit
def tropical_min_plus_par(a, b):
  """One min-plus matrix product: C[i, j] = min_r (A[i, r] + B[r, j])."""
  m, k = a.shape
  _, n = b.shape
  return par.imap(
    lambda i, j: par.ireduce(lambda r: a[i, r] + b[r, j], k, combine=np.minimum),
    (m, n),
  )

    4. Sequential recurrences to parallel scans¶

    Exponential IIR filter¶

    A plain for t in range(T): affine recurrence s = a[t] * s + b[t] works for arbitrary a (including zero or negative coefficients) without numerical underflow and rewrites via Law 4 chunkwise duality into a division-free parallel prefix scan. For short sequences with strictly positive decays (a > 0), the closed-form prefix product can also be written directly in NumPy or par.scan (note that b / np.cumprod(a) underflows in float32 for long sequences, and _comp does O(T²) work):

    @par.jit
def iir_filter_loop(a, b):
  # Works for arbitrary a (including 0 or negative); Law 4 rewrites to TileScan.
  T = a.shape[0]
  out = np.empty_like(b)
  s = np.float32(0.0)
  for t in range(T):
    s = a[t] * s + b[t]
    out[t] = s
  return out

    @par.jit
def iir_filter_numpy(a, b):
  # Closed-form prefix formula for strictly positive decay coefficients a > 0.
  p = np.cumprod(a)
  return p * np.cumsum(b / p)

    @par.jit
def iir_filter_comp(a, b):
  p = np.cumprod(a)
  scaled = b / p
  return np.array([p[t] * np.sum(scaled[: t + 1]) for t in range(len(a))])

    @par.jit
def iir_filter_par(a, b):
  p = par.scan(operator.mul, a)
  return p * par.scan(operator.add, b / p)

    Mamba-2 state-space recurrence¶

    The same Law 4 pass turns a 6-line bilinear matrix-state loop into division-free chunkwise GEMMs that lower directly to CPU register tiles, Metal simdgroup_matrix, and CUDA TF32 WMMA:

    @par.jit
def mamba2_ssd(q, k, v, a):
  T = q.shape[0]
  out = np.empty((T, v.shape[1]), dtype=np.float32)
  S = np.zeros((k.shape[1], v.shape[1]), dtype=np.float32)
  for t in range(T):
    S = a[t] * S + k[t][:, None] * v[t][None, :]
    out[t] = q[t] @ S
  return out

    5. Attention and large-vocabulary losses¶

    Causal attention¶

    The _numpy and _helpers (par.nn.scaled_dot_product_attention) forms automatically compose online monoidal tiling (no T × T matrices), tile-interval block skipping (upper triangle never loaded), and rematerialized AD under par.grad, par.jvp, and par.hvp. The _comp, _loop, and _par forms compute the exact same result without triggering flash_monoid:

    @par.jit
def causal_attention_numpy(q, k, v):
  T = q.shape[-2]
  idx = np.arange(T)
  mask = idx[:, None] >= idx[None, :]
  scale = np.float32(1.0 / np.sqrt(np.float32(q.shape[-1])))
  s = np.where(mask, np.matmul(q, np.swapaxes(k, -1, -2)) * scale, np.float32(-1e9))
  e = np.exp(s - np.max(s, axis=-1, keepdims=True))
  return np.matmul(e / np.sum(e, axis=-1, keepdims=True), v)

    @par.jit
def causal_attention_helpers(q, k, v):
  return par.nn.scaled_dot_product_attention(q, k, v, causal=True)

    @par.jit
def causal_attention_comp(q, k, v):
  T = q.shape[-2]
  idx = np.arange(T)
  mask = idx[:, None] >= idx[None, :]
  scale = np.float32(1.0 / np.sqrt(np.float32(q.shape[-1])))

  def one_head(qh, kh, vh):
    s = np.where(mask, (qh @ kh.T) * scale, np.float32(-1e9))
    e = np.exp(s - np.max(s, axis=-1, keepdims=True))
    return (e / np.sum(e, axis=-1, keepdims=True)) @ vh

  return np.array([[one_head(q[b, h], k[b, h], v[b, h]) for h in range(q.shape[1])] for b in range(q.shape[0])])

    @par.jit
def causal_attention_loop(q, k, v):
  B, H, T, D = q.shape
  scale = np.float32(1.0 / np.sqrt(np.float32(D)))
  out = np.zeros((B, H, T, v.shape[-1]), dtype=q.dtype)
  for b in range(B):
    for h in range(H):
      for i in range(T):
        m = -np.inf
        for j in range(i + 1):
          score = np.sum(q[b, h, i] * k[b, h, j]) * scale
          if score > m:
            m = score
        denom = 0.0
        for j in range(i + 1):
          w = np.exp(np.sum(q[b, h, i] * k[b, h, j]) * scale - m)
          denom += w
          out[b, h, i] += w * v[b, h, j]
        out[b, h, i] /= denom
  return out

    @par.jit
def causal_attention_par(q, k, v):
  T = q.shape[-2]
  idx = np.arange(T)
  mask = idx[:, None] >= idx[None, :]
  scale = np.float32(1.0 / np.sqrt(np.float32(q.shape[-1])))

  def one_head(qh, kh, vh):
    s = np.where(mask, (qh @ kh.T) * scale, np.float32(-1e9))
    e = np.exp(s - np.max(s, axis=-1, keepdims=True))
    return (e / np.sum(e, axis=-1, keepdims=True)) @ vh

  return par.map(lambda qb, kb, vb: par.map(one_head, qb, kb, vb), q, k, v)

    Intersect the mask with (idx[:, None] - idx[None, :]) <= window for an automatic O(N · W) sliding window.

    Linear cross-entropy¶

    The _numpy and _helpers (par.nn.cross_entropy(h @ w, targets)) forms tile h @ w and logsumexp over the vocabulary V without allocating an M × V logit matrix (forward and backward); _comp, _loop, and _par show the equivalent row-wise spellings:

    @par.jit
def cross_entropy_numpy(h, w, targets):
  logits = h @ w
  m = np.max(logits, axis=-1, keepdims=True)
  z = logits - m
  lse = np.log(np.sum(np.exp(z), axis=-1))
  picked = z[np.arange(h.shape[0]), targets]
  return -np.mean(picked - lse)

    @par.jit
def cross_entropy_helpers(h, w, targets):
  return par.nn.cross_entropy(h @ w, targets)

    @par.jit
def cross_entropy_comp(h, w, targets):
  def row_loss(row, tgt):
    logits = row @ w
    z = logits - np.max(logits)
    return np.log(np.sum(np.exp(z))) - z[tgt]

  return np.mean(np.array([row_loss(row, tgt) for row, tgt in zip(h, targets)]))

    @par.jit
def cross_entropy_loop(h, w, targets):
  total = 0.0
  for i in range(len(targets)):
    logits = h[i] @ w
    z = logits - np.max(logits)
    total += np.log(np.sum(np.exp(z))) - z[targets[i]]
  return total / len(targets)

    @par.jit
def cross_entropy_par(h, w, targets):
  def row_loss(row, tgt):
    logits = row @ w
    z = logits - np.max(logits)
    return np.log(np.sum(np.exp(z))) - z[tgt]

  return np.mean(par.map(row_loss, h, targets))

    Multi-head latent attention¶

    Law 3 contraction reassociation absorbs w_uk into q and defers w_uv during short-query decode so tiled attention runs directly over the compressed latent c_kv:

    @par.jit
def deepseek_mla(q, c_kv, w_uk, w_uv):
  k = c_kv @ w_uk
  v = c_kv @ w_uv
  scale = np.float32(1.0 / np.sqrt(np.float32(q.shape[-1])))
  s = (q @ k.T) * scale
  e = np.exp(s - np.max(s, axis=-1, keepdims=True))
  return (e / np.sum(e, axis=-1, keepdims=True)) @ v

    Grouped-query attention¶

    Shared-KV head-group tiling folds the head-group axis G into the query tile so all G query heads reuse one shared-memory load of k and v:

    @par.jit
def gqa_attention(q, k, v):
  # q: (B, H_kv, G, T_q, D), k, v: (B, H_kv, 1, T_kv, D)
  scale = np.float32(1.0 / np.sqrt(np.float32(q.shape[-1])))
  s = np.matmul(q, np.swapaxes(k, -1, -2)) * scale
  e = np.exp(s - np.max(s, axis=-1, keepdims=True))
  return np.matmul(e / np.sum(e, axis=-1, keepdims=True), v)

    6. Objects, methods, and full programs¶

    Dataclasses, methods, and polymorphic dispatch¶

    @dataclass types and par.nn.Module subclasses inline pure methods, @property, __call__, and structural match/case completely during elaboration with zero PyObject overhead:

    @dataclass
class ProteinBackbone:
  """Mutable-style @dataclass with pure methods, @property, and functional field updates."""

  coords: np.ndarray  # (N, 3, 3) for N, CA, C atoms per residue
  residue_ids: np.ndarray  # (N,) int64 metadata (zero-tangent static leaf under AD)
  chain_name: str = "A"

  @property
  def ca_coords(self):
    return self.coords[:, 1, :]

  @property
  def n_residues(self):
    return self.coords.shape[0]

  def centered(self):
    mu = np.mean(self.ca_coords, axis=0, keepdims=True)
    self.coords = self.coords - mu[:, None, :]
    return self

  def rbf_message_passing(self, features, neg_gamma):
    ca = self.ca_coords
    d2 = np.sum((ca[:, None, :] - ca[None, :, :]) ** np.float32(2.0), axis=-1)
    s = d2 * neg_gamma
    e = np.exp(s - np.max(s, axis=-1, keepdims=True))
    return (e / np.sum(e, axis=-1, keepdims=True)) @ features


@dataclass
class MLPBlock:
  """Callable @dataclass layer with polymorphic dense/ragged dispatch."""

  w1: np.ndarray
  b1: np.ndarray
  w2: np.ndarray
  b2: np.ndarray
  name: str = "mlp"

  def __call__(self, x, *, mask=None, scale=1.0):
    if isinstance(x, RaggedArray):
      h = np.tanh(x.values @ self.w1 + self.b1)
      y = (h @ self.w2 + self.b2) * scale
      return RaggedArray(y, x.offsets)
    h = np.tanh(x @ self.w1 + self.b1)
    y = (h @ self.w2 + self.b2) * scale
    if mask is not None:
      y = np.where(mask[:, None], y, np.float32(0.0))
    return y


def apply_with_kwargs(fn, x, *args, **kwargs):
  return fn(x, *args, **kwargs)


@par.jit
def score_structure(backbone, features, block, neg_gamma, mask=None):
  bb = backbone.centered()
  msgs = bb.rbf_message_passing(features, neg_gamma)
  step = functools.partial(apply_with_kwargs, block, scale=np.float32(0.5))
  out = step(msgs, mask=mask)
  match out:
    case RaggedArray(values=vals):
      return par.reduce(operator.add, vals * vals)
    case _:
      return np.sum(out * out)
