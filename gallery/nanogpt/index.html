    Train nanoGPT in Parakeet¶

    This character-level GPT learns to predict the next character in a short passage. For the fragment para, one training sequence has inputs p a r and targets a r a. A lower-triangular causal mask ensures position t only attends to positions 0..t while computing every position's logits in parallel.

    Below we show two complete implementations of the exact same model:

    1. Pythonic objects with par.nn and par.optim (examples/nanogpt.py): a normal Python class inheriting from par.nn.Module with a standard __init__, par.nn.TransformerBlock, par.nn.cross_entropy, par.optim.Adam, and par.Vocab.from_text.
    2. Minimalist functional from scratch (examples/nanogpt_functional.py): plain dicts and tuples of NumPy arrays with every layer (layer norm, softmax, multi-head attention, cross-entropy, and Adam) written inline in raw NumPy—no par.nn or par.optim helpers at all.

    Run either version directly from a checkout:

    python -m examples.nanogpt
python -m examples.nanogpt_functional

    How par.nn.Module compiles Python objects¶

    Parakeet does not require external init() factories or manual dataclass unpacking to compile neural networks. Any class inheriting from par.nn.Module (or setting __parakeet_record__ = True) is lowered at @par.jit and par.value_and_grad boundaries into a flat compile-time record of its self.__dict__ attributes:

    • Array attributes (weight, bias, gain, wpe) become native typed array parameters that receive gradients under par.value_and_grad, transfer cleanly with par.to_device, and update via par.tree_map_arrays / par.optim.
    • Static attributes (heads, eps, activation) become compile-time constants—zero runtime memory and untouched by optimizers.
    • Methods (__call__, helper methods) inline completely during elaboration with zero PyObject overhead.

    Because every layer in par.nn (Linear, LayerNorm, Embedding, MLP, CausalSelfAttention, TransformerBlock) is just plain NumPy under the hood, Parakeet's monoidal FlashAttention tiling, warp-row LayerNorm reduction, and rematerialized AD trigger automatically on both versions.

    Model definition¶

    """Character-level nanoGPT using ``par.nn.Module``, ``par.optim.Adam``, and ``par.Vocab``.

``NanoGPT`` is a normal Python class with a standard ``__init__``. Because it
inherits from :class:`par.nn.Module`, Parakeet lowers the model and its methods
into compiled native code with zero ``PyObject`` overhead, differentiates the
loss with ``par.value_and_grad``, and updates every parameter array with
``par.optim.Adam``.

For a zero-helper functional version written with plain dicts and raw NumPy,
see ``examples/nanogpt_functional.py``.
"""

from __future__ import annotations

import time

import numpy as np

import parakeet as par


class NanoGPT(par.nn.Module):
  """Character-level transformer with standard Python ``__init__``."""

  def __init__(self, vocab: int, context: int, width: int, layers: int, *, heads: int = 2, rng=None, dtype=np.float64):
    g = np.random.default_rng(0) if rng is None else rng
    self.wte = par.nn.Embedding(vocab, width, rng=g, dtype=dtype)
    self.wpe = g.normal(0.0, 0.02, (context, width)).astype(dtype)
    self.blocks = tuple(par.nn.TransformerBlock(width, heads, rng=g, dtype=dtype) for _ in range(layers))
    self.ln_f = par.nn.LayerNorm(width, dtype=dtype)
    self.head = par.nn.Linear(width, vocab, bias=False, rng=g, dtype=dtype, std=0.02)

  def __call__(self, tokens, mask=None):
    x = self.wte(tokens) + self.wpe
    for block in self.blocks:
      x = block(x, mask=mask)
    return self.head(self.ln_f(x))


def loss(model, tokens, targets, mask):
  return par.nn.cross_entropy(model(tokens, mask=mask), targets)


loss_and_grad = par.value_and_grad(loss)

    """Minimalist functional nanoGPT written from scratch without ``par.nn`` or ``par.optim``.

Parameters live in plain dicts and tuples of NumPy arrays. Every neural-network
building block (layer normalization, softmax, multi-head causal attention,
cross-entropy, and Adam) is written inline in a few lines of NumPy so there are
no hidden abstractions.

Compare with the Pythonic ``par.nn.Module`` + ``par.optim.Adam`` version in
``examples/nanogpt.py``.
"""

from __future__ import annotations

import time

import numpy as np

import parakeet as par

N_HEAD = 2


def layer_norm(x, gain, bias, eps=1e-5):
  mu = np.mean(x, axis=-1, keepdims=True)
  var = np.mean((x - mu) ** 2, axis=-1, keepdims=True)
  return (x - mu) / np.sqrt(var + eps) * gain + bias


def softmax(x):
  shifted = x - np.max(x, axis=-1, keepdims=True)
  exps = np.exp(shifted)
  return exps / np.sum(exps, axis=-1, keepdims=True)


def causal_attention(x, block, mask):
  """Multi-head causal self-attention over ``(B, T, C)``."""
  b, t, c = x.shape
  d = c // N_HEAD
  q = (x @ block["wq"]).reshape((b, t, N_HEAD, d)).transpose((0, 2, 1, 3))
  k = (x @ block["wk"]).reshape((b, t, N_HEAD, d)).transpose((0, 2, 1, 3))
  v = (x @ block["wv"]).reshape((b, t, N_HEAD, d)).transpose((0, 2, 1, 3))
  scores = np.matmul(q, np.swapaxes(k, -1, -2)) / np.sqrt(d)
  weights = softmax(np.where(mask, scores, -1e9))
  out = np.matmul(weights, v).transpose((0, 2, 1, 3)).reshape((b, t, c))
  return out @ block["wo"]


def transformer_block(x, block, mask):
  x = x + causal_attention(layer_norm(x, block["g1"], block["b1"]), block, mask)
  h = np.tanh(layer_norm(x, block["g2"], block["b2"]) @ block["w1"] + block["bf1"])
  return x + h @ block["w2"] + block["bf2"]

    def loss(params, tokens, targets, mask):
  b, t = tokens.shape
  c = params["wte"].shape[1]
  x = params["wte"][tokens.reshape((b * t,))].reshape((b, t, c)) + params["wpe"]
  for block in params["blocks"]:
    x = transformer_block(x, block, mask)
  logits = (layer_norm(x, params["gf"], params["bf"]) @ params["head"]).reshape((b * t, params["head"].shape[1]))
  shifted = logits - np.max(logits, axis=1, keepdims=True)
  log_denom = np.log(np.sum(np.exp(shifted), axis=1))
  picked = shifted[np.arange(b * t), targets.reshape((b * t,))]
  return -np.mean(picked - log_denom)


loss_and_grad = par.value_and_grad(loss)


def init_params(vocab, context, width, layers, rng):
  w = lambda *shape: rng.normal(0.0, 0.02, shape)  # noqa: E731
  blocks = tuple(
    {
      "g1": np.ones(width),
      "b1": np.zeros(width),
      "wq": w(width, width),
      "wk": w(width, width),
      "wv": w(width, width),
      "wo": w(width, width),
      "g2": np.ones(width),
      "b2": np.zeros(width),
      "w1": w(width, 4 * width),
      "bf1": np.zeros(4 * width),
      "w2": w(4 * width, width),
      "bf2": np.zeros(width),
    }
    for _ in range(layers)
  )
  return {
    "wte": w(vocab, width),
    "wpe": w(context, width),
    "blocks": blocks,
    "gf": np.ones(width),
    "bf": np.zeros(width),
    "head": w(width, vocab),
  }


def adam_step(params, grads, m, v, step, lr=3e-3, b1=0.9, b2=0.99):
  m = par.tree_map(lambda mi, g: b1 * mi + (1.0 - b1) * g, m, grads)
  v = par.tree_map(lambda vi, g: b2 * vi + (1.0 - b2) * g * g, v, grads)
  scale = lr * np.sqrt(1.0 - b2**step) / (1.0 - b1**step)
  params = par.tree_map(lambda p, mi, vi: p - scale * mi / (np.sqrt(vi) + 1e-8), params, m, v)
  return params, m, v

    Tokenization and training loop¶

    par.Vocab.from_text(TEXT) builds a compact character vocabulary backed by a 256-entry lookup table—the exact same categorical encoding pattern as par.Alphabet("ACGT") in the bioinformatics gallery. Calling vocab.encode(TEXT) returns a 1D int64 token array ready for windowed batching.

def sample_batch(data, batch: int, context: int, rng):
  starts = rng.integers(0, len(data) - context - 1, batch)
  tokens = np.stack([data[i : i + context] for i in starts])
  targets = np.stack([data[i + 1 : i + context + 1] for i in starts])
  return tokens, targets


def main(steps=60, batch=16, context=32, width=32, layers=2, heads=2, seed=0, verbose=False):
  rng = np.random.default_rng(seed)
  vocab = par.Vocab.from_text(TEXT)
  data = vocab.encode(TEXT)

  model = NanoGPT(len(vocab), context, width, layers, heads=heads, rng=rng)
  optimizer = par.optim.Adam(model, lr=3e-3, betas=(0.9, 0.99))
  mask = par.nn.causal_mask(context)

  first = last = None
  start = time.perf_counter()
  for step in range(1, steps + 1):
    tokens, targets = sample_batch(data, batch, context, rng)
    value, grads = loss_and_grad(model, tokens, targets, mask)
    model = optimizer.update(model, grads)
    first = value if first is None else first
    last = value
    if verbose and (step == 1 or step % 20 == 0):
      print(f"step {step:3d}  loss {value:.3f}  ({time.perf_counter() - start:.1f}s)")

  assert last < 0.8 * first, (first, last)
  return first, last


if __name__ == "__main__":
  main(verbose=True)

    Fusing the optimizer step on GPU¶

    In the walkthrough above, optimizer.update(model, grads) runs in Python so each step is easy to inspect. When training on a GPU, wrap both par.value_and_grad(loss) and par.optim.adam_step inside one @par.jit function and keep (model, m, v) resident on the device with par.to_device:

    @par.jit(backend="cuda", tensor_cores=True)
def train_step(model, m, v, step, tokens, targets, mask):
  value, grads = par.value_and_grad(loss)(model, tokens, targets, mask)
  model, m, v = par.optim.adam_step(model, grads, m, v, step, lr=3e-3)
  return value, model, m, v

    That compiles forward, backward, and Adam into a single zero-round-trip device step. See the nanoGPT GPU benchmarks for full timings comparing Parakeet FP32/TF32, PyTorch (eager, SDPA, torch.compile), and JAX.
